```python
# src/sus_py/rules.py
"""Security rules for different strictness levels."""

STRICT_RULES = {
    'dangerous_imports': ['os', 'subprocess', 'shutil', 'sys', 'socket'],
    'dangerous_calls': ['eval', 'exec', 'compile', '__import__']
}

MODERATE_RULES = {
    'dangerous_imports': ['subprocess', 'shutil'],
    'dangerous_calls': ['eval', 'exec']
}

LOOSE_RULES = {
    'dangerous_calls': ['eval', 'exec']
}

def get_rules(level: str) -> dict:
    """Get rules for the specified strictness level."""
    rules_map = {
        'strict': STRICT_RULES,
        'moderate': MODERATE_RULES,
        'loose': LOOSE_RULES
    }
    return rules_map.get(level, MODERATE_RULES)
```

```python
# src/sus_py/analyzer.py
"""AST-based security analyzer."""
import ast
from typing import List, Dict, Any

class SecurityVisitor(ast.NodeVisitor):
    """AST visitor that detects dangerous imports and calls."""

    def __init__(self, rules: dict):
        self.rules = rules
        self.issues: List[Dict[str, Any]] = []

    def visit_Import(self, node: ast.Import):
        """Check for dangerous imports like 'import os'."""
        for alias in node.names:
            if alias.name in self.rules.get('dangerous_imports', []):
                self.issues.append({
                    'line': node.lineno,
                    'type': 'import',
                    'name': alias.name,
                    'severity': 'CRITICAL'
                })
        self.generic_visit(node)

    def visit_ImportFrom(self, node: ast.ImportFrom):
        """Check for dangerous from imports like 'from os import system'."""
        if node.module and node.module in self.rules.get('dangerous_imports', []):
            self.issues.append({
                'line': node.lineno,
                'type': 'import',
                'name': node.module,
                'severity': 'CRITICAL'
            })
        self.generic_visit(node)

    def visit_Call(self, node: ast.Call):
        """Check for dangerous function calls like 'eval()'."""
        if isinstance(node.func, ast.Name):
            if node.func.id in self.rules.get('dangerous_calls', []):
                self.issues.append({
                    'line': node.lineno,
                    'type': 'call',
                    'name': node.func.id,
                    'severity': 'HIGH'
                })
        self.generic_visit(node)

def analyze_code(code: str, rules: dict) -> List[Dict[str, Any]]:
    """Analyze Python code and return list of security issues."""
    try:
        tree = ast.parse(code)
        visitor = SecurityVisitor(rules)
        visitor.visit(tree)
        return visitor.issues
    except SyntaxError:
        raise
```

```python
"""Report formatting using rich."""
from typing import List, Dict, Any
from rich.console import Console
from rich.table import Table

def format_report(issues: List[Dict[str, Any]], filename: str) -> str:
    """Format security issues as a rich report."""
    console = Console()

    if not issues:
        return "✅ All clear - No security issues detected"

    table = Table(title=f"🚨 Security Issues in {filename}")
    table.add_column("Line", style="cyan")
    table.add_column("Type", style="magenta")
    table.add_column("Name", style="red")
    table.add_column("Severity", style="yellow")

    for issue in issues:
        table.add_row(
            str(issue['line']),
            issue['type'],
            issue['name'],
            issue['severity']
        )

    # Capture table output
    from io import StringIO
    buffer = StringIO()
    temp_console = Console(file=buffer, force_terminal=True)
    temp_console.print(table)
    return buffer.getvalue()
```

```python
# src/sus_py/cli.py
"""CLI interface using typer."""
import sys
from pathlib import Path
import typer
from rich.console import Console

from sus_py.analyzer import analyze_code
from sus_py.rules import get_rules
from sus_py.reporter import format_report

app = typer.Typer()
console = Console()

@app.command()
def main(
    file_path: Path = typer.Argument(..., help="Python file to scan"),
    strict: bool = typer.Option(False, "--strict", help="Use strict mode"),
    loose: bool = typer.Option(False, "--loose", help="Use loose mode"),
):
    """Scan Python file for suspicious code."""

    # Determine strictness level
    if strict:
        level = 'strict'
    elif loose:
        level = 'loose'
    else:
        level = 'moderate'

    # Check file exists
    if not file_path.exists():
        console.print(f"[red]Error: File not found: {file_path}[/red]")
        sys.exit(2)

    # Read and analyze
    try:
        code = file_path.read_text(encoding='utf-8')
        rules = get_rules(level)
        issues = analyze_code(code, rules)

        # Print report
        report = format_report(issues, file_path.name)
        console.print(report)

        # Exit with appropriate code
        sys.exit(1 if issues else 0)

    except SyntaxError as e:
        console.print(f"[red]Parse Error: Invalid Python syntax at line {e.lineno}[/red]")
        sys.exit(2)
    except Exception as e:
        console.print(f"[red]Error: {e}[/red]")
        sys.exit(2)

if __name__ == "__main__":
    app()
```

```bash
# Scan a file (moderate mode by default)
sus script.py

sus --strict script.py

sus --loose script.py
```

## Exit Codes

- `0`: No security issues detected
- `1`: Security issues found
- `2`: Parse error or file not found

**Strict Mode:**
- Dangerous imports: `os`, `subprocess`, `shutil`, `sys`, `socket`
- Dangerous calls: `eval`, `exec`, `compile`, `__import__`

**Moderate Mode (default):**
- Dangerous imports: `subprocess`, `shutil`
- Dangerous calls: `eval`, `exec`

**Loose Mode:**
- Dangerous calls: `eval`, `exec`
